# Frontend (Angular + PrimeNG) + wdrożenie — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publiczna strona przedszkola (strona główna z kafelkami, podstrony, pasek dostępności, motywy sezonowe) i panel admina (wpisy ze zdjęciami, grupy, treści, motywy, użytkownicy) oraz paczka do wgrania na OVH przez FileZillę.

**Architecture:** Angular (standalone components, signals, control flow `@if/@for`), PrimeNG z presetem Aura i paletą primary podmienianą w runtime z `GET /api/theme`. Publiczne trasy prerenderowane w czasie builda (`@angular/ssr`, `outputMode: static`), panel `/admin` ładowany leniwie i renderowany po stronie klienta. Zdjęcia zmniejszane w przeglądarce (Web Worker + OffscreenCanvas) i wysyłane kolejką po 3.

**Tech Stack:** Angular (aktualna stabilna, ≥ 20), PrimeNG (wersja zgodna z Angular) + `@primeuix/themes`, PrimeIcons, `@angular/cdk` (drag & drop), Quill (dla `p-editor`), `@fontsource/atkinson-hyperlegible`, Vitest (domyślny runner `ng test`), Playwright + `@axe-core/playwright`.

**Spec:** `docs/superpowers/specs/2026-10-15-przedszkole-design.md`
**Kontrakt API:** `docs/api.md` (Task B0 w `docs/superpowers/plans/2026-10-15-backend.md`)

## Review Focus

1. **Zdjęcia HEIC/nieobsługiwane z iPhone'a** — `createImageBitmap` rzuca błąd; ten plik dostaje status „błąd” z czytelnym komunikatem, pozostałe pliki z tej samej partii wysyłają się dalej. Test w Task F11 (`UploadQueue`) i F12 (komunikat).
2. **Wygaśnięcie tokenu w trakcie wysyłania 30 zdjęć** — 3 równoległe żądania dostają 401 naraz; ma pójść **jedno** `/auth/refresh`, a potem wszystkie 3 są ponowione. Test w Task F9.
3. **Safari bez kodowania WebP** — `convertToBlob({type:'image/webp'})` zwraca PNG; trzeba wtedy zakodować JPEG (backend odrzuca PNG). Test w Task F11 (`encodeWithFallback`).
4. **Uszkodzony / stary stan w `localStorage`** (np. ręcznie zmieniony, inna wersja) — pasek dostępności wraca do ustawień domyślnych zamiast wywracać aplikację; tryb prywatny Safari (wyjątek przy `setItem`) nie psuje przełączników. Test w Task F4.
5. **Brak API przy prerenderze/awaria API** — build prerenderuje strony bez backendu, a gdy `/api/theme` nie odpowiada, strona i tak ma kolory pory roku liczonej lokalnie (strefa Warszawy). Testy w Task F3 i F18.

import { Palette, Shade, Theme } from '../core/models';

export const SHADES: readonly Shade[] = ['50', '100', '200', '300', '400', '500', '600', '700', '800', '900'];

const palette = (colors: string[]): Palette =>
  ({ ...Object.fromEntries(SHADES.map((k, i) => [k, colors[i]])), accent: colors[7], text: colors[9] }) as Palette;

const season = (key: string, name: string, seasonStart: string, colors: string[]): Theme =>
  ({ id: null, key, name, type: 'season', seasonStart, palette: palette(colors) });

/** Te same wartości co backend/src/Theme/DefaultThemes.php. */
export const DEFAULT_THEMES: Theme[] = [
  season('spring', 'Wiosna', '03-21', ['#f0fdf4', '#dcfce7', '#bbf7d0', '#86efac', '#4ade80', '#22c55e', '#16a34a', '#15803d', '#166534', '#14532d']),
  season('summer', 'Lato', '06-23', ['#fefce8', '#fef9c3', '#fef08a', '#fde047', '#facc15', '#eab308', '#ca8a04', '#a16207', '#854d0e', '#713f12']),
  season('autumn', 'Jesień', '09-23', ['#fef2f2', '#fee2e2', '#fecaca', '#fca5a5', '#f87171', '#ef4444', '#dc2626', '#b91c1c', '#991b1b', '#7f1d1d']),
  season('winter', 'Zima', '12-21', ['#eff6ff', '#dbeafe', '#bfdbfe', '#93c5fd', '#60a5fa', '#3b82f6', '#2563eb', '#1d4ed8', '#1e40af', '#1e3a8a']),
];

export function seasonFor(date: Date, starts: Record<string, string>): string {
  const parts = new Intl.DateTimeFormat('en-GB', { timeZone: 'Europe/Warsaw', month: '2-digit', day: '2-digit' }).formatToParts(date);
  const md = `${parts.find((p) => p.type === 'month')!.value}-${parts.find((p) => p.type === 'day')!.value}`;
  const sorted = Object.entries(starts).sort(([, a], [, b]) => a.localeCompare(b));
  let current = sorted[sorted.length - 1][0];
  for (const [key, start] of sorted) {
    if (md >= start) current = key;
  }
  return current;
}

export function fallbackTheme(date: Date): Theme {
  const starts = Object.fromEntries(DEFAULT_THEMES.map((t) => [t.key, t.seasonStart!]));
  const key = seasonFor(date, starts);
  return DEFAULT_THEMES.find((t) => t.key === key)!;
}

import { DOCUMENT, isPlatformBrowser } from '@angular/common';
import { effect, inject, Injectable, PLATFORM_ID, signal } from '@angular/core';

export interface A11yState { fontScale: 0 | 1 | 2; contrast: boolean; readableFont: boolean; grayscale: boolean }
type ToggleKey = 'contrast' | 'readableFont' | 'grayscale';

const KEY = 'a11y';
const DEFAULT: A11yState = { fontScale: 0, contrast: false, readableFont: false, grayscale: false };

@Injectable({ providedIn: 'root' })
export class A11yService {
  private readonly doc = inject(DOCUMENT);
  private readonly browser = isPlatformBrowser(inject(PLATFORM_ID));
  readonly state = signal<A11yState>(this.restore());

  constructor() {
    effect(() => this.applyAndPersist(this.state()));
  }

  increaseFont(): void {
    this.state.update((s) => ({ ...s, fontScale: Math.min(2, s.fontScale + 1) as A11yState['fontScale'] }));
  }

  decreaseFont(): void {
    this.state.update((s) => ({ ...s, fontScale: Math.max(0, s.fontScale - 1) as A11yState['fontScale'] }));
  }

  resetFont(): void {
    this.state.update((s) => ({ ...s, fontScale: 0 }));
  }

  toggle(key: ToggleKey): void {
    this.state.update((s) => ({ ...s, [key]: !s[key] }));
  }

  private restore(): A11yState {
    if (!this.browser) return DEFAULT;
    try {
      const raw = JSON.parse(localStorage.getItem(KEY) ?? 'null') as Partial<A11yState> | null;
      if (!raw || typeof raw !== 'object') return DEFAULT;
      return {
        fontScale: raw.fontScale === 1 || raw.fontScale === 2 ? raw.fontScale : 0,
        contrast: raw.contrast === true,
        readableFont: raw.readableFont === true,
        grayscale: raw.grayscale === true,
      };
    } catch {
      return DEFAULT;
    }
  }

  private applyAndPersist(s: A11yState): void {
    const c = this.doc.documentElement.classList;
    c.toggle('a11y-font-1', s.fontScale === 1);
    c.toggle('a11y-font-2', s.fontScale === 2);
    c.toggle('a11y-contrast', s.contrast);
    c.toggle('a11y-readable', s.readableFont);
    c.toggle('a11y-grayscale', s.grayscale);
    if (!this.browser) return;
    try {
      localStorage.setItem(KEY, JSON.stringify(s));
    } catch {
      /* tryb prywatny / brak miejsca — ustawienia działają do końca wizyty */
    }
  }
}

import { DOCUMENT } from '@angular/common';
import { inject, Injectable, signal } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class SpeechService {
  private readonly win = inject(DOCUMENT).defaultView;
  readonly supported = !!this.win && 'speechSynthesis' in this.win && 'SpeechSynthesisUtterance' in this.win;
  readonly speaking = signal(false);

  speak(text: string): void {
    if (!this.supported || !text.trim()) return;
    const synth = this.win!.speechSynthesis;
    synth.cancel();
    const u = new SpeechSynthesisUtterance(text.trim());
    u.lang = 'pl-PL';
    const voice = synth.getVoices().find((v) => v.lang.toLowerCase().startsWith('pl'));
    if (voice) u.voice = voice;
    u.onend = u.onerror = () => this.speaking.set(false);
    this.speaking.set(true);
    synth.speak(u);
  }

  stop(): void {

export function formatSize(bytes: number): string {

import { HttpClient } from '@angular/common/http';
import { computed, inject, Injectable, signal } from '@angular/core';
import { finalize, firstValueFrom, map, Observable, shareReplay, tap } from 'rxjs';
import { API } from '../config';
import { AuthUser, LoginResponse } from '../models';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly http = inject(HttpClient);
  private token: string | null = null;
  private refreshInFlight: Observable<string> | null = null;
  readonly user = signal<AuthUser | null>(null);
  readonly isAdmin = computed(() => this.user()?.role === 'admin');

  accessToken(): string | null {
    return this.token;
  }

  async login(email: string, password: string): Promise<AuthUser> {
    const res = await firstValueFrom(this.http.post<LoginResponse>(`${API}/auth/login`, { email, password }));
    this.setSession(res);
    return res.user;
  }

  /** Jedno żądanie /auth/refresh współdzielone przez wszystkie równoległe wywołania. */
  refresh(): Observable<string> {
    this.refreshInFlight ??= this.http.post<LoginResponse>(`${API}/auth/refresh`, {}).pipe(
      tap((res) => this.setSession(res)),
      map((res) => res.accessToken),
      finalize(() => (this.refreshInFlight = null)),
      shareReplay(1),
    );
    return this.refreshInFlight;
  }

  async ensureSession(): Promise<boolean> {
    if (this.token) return true;
    try {
      await firstValueFrom(this.refresh());
      return true;
    } catch {
      return false;
    }
  }

  async logout(): Promise<void> {
    try {
      await firstValueFrom(this.http.post(`${API}/auth/logout`, {}));
    } finally {
      this.clear();
    }
  }

  clear(): void {
    this.token = null;
    this.user.set(null);
  }

  private setSession(res: LoginResponse): void {
    this.token = res.accessToken;
    this.user.set(res.user);
  }
}

import { HttpErrorResponse, HttpInterceptorFn, HttpRequest } from '@angular/common/http';
import { inject } from '@angular/core';
import { Router } from '@angular/router';
import { catchError, switchMap, throwError } from 'rxjs';
import { API } from '../config';
import { AuthService } from './auth.service';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  if (!req.url.startsWith(`${API}/admin`)) return next(req);
  const auth = inject(AuthService);
  const router = inject(Router);
  const withToken = (r: HttpRequest<unknown>, token: string | null) =>
    token ? r.clone({ setHeaders: { Authorization: `Bearer ${token}` } }) : r;

  return next(withToken(req, auth.accessToken())).pipe(
    catchError((err: unknown) => {
      if (!(err instanceof HttpErrorResponse) || err.status !== 401) return throwError(() => err);
      return auth.refresh().pipe(
        catchError((refreshErr: unknown) => {
          auth.clear();
          void router.navigate(['/admin/logowanie']);
          return throwError(() => refreshErr);
        }),
        // ponowienie idzie przez `next`, więc drugie 401 nie wywoła kolejnego refresh
        switchMap((token) => next(withToken(req, token))),
      );
    }),
  );
};

export const THUMB_SIDE = 400;
export const QUALITY = 0.8;

export class UnsupportedImageError extends Error {
  constructor() {
    super('Nieobsługiwany format zdjęcia (np. HEIC). Ustaw w telefonie zapis zdjęć jako JPG lub wybierz inne zdjęcie.');
  }

export function targetSize(w: number, h: number, max: number): { w: number; h: number } {
  const scale = Math.min(1, max / Math.max(w, h));
  return { w: Math.round(w * scale), h: Math.round(h * scale) };
}

/** Safari nie koduje WebP z canvasa i po cichu zwraca PNG — wtedy kodujemy JPEG. */
export async function encodeWithFallback(encode: (type: string, quality: number) => Promise<Blob>, quality: number): Promise<Blob> {
  const webp = await encode('image/webp', quality);
  return webp.type === 'image/webp' ? webp : encode('image/jpeg', quality);
}

type AnyCanvas = OffscreenCanvas | HTMLCanvasElement;

function makeCanvas(w: number, h: number): AnyCanvas {
  if (typeof OffscreenCanvas !== 'undefined') return new OffscreenCanvas(w, h);
  const c = document.createElement('canvas');
  c.width = w;
  c.height = h;
  return c;
}

function encoder(canvas: AnyCanvas) {
  return (type: string, quality: number): Promise<Blob> =>
    canvas instanceof HTMLCanvasElement
      ? new Promise((resolve, reject) => canvas.toBlob((b) => (b ? resolve(b) : reject(new Error('toBlob'))), type, quality))
      : canvas.convertToBlob({ type, quality });
}

async function render(bitmap: ImageBitmap, max: number): Promise<Blob> {
  const size = targetSize(bitmap.width, bitmap.height, max);
  const canvas = makeCanvas(size.w, size.h);
  const ctx = canvas.getContext('2d') as CanvasRenderingContext2D | OffscreenCanvasRenderingContext2D;
  ctx.imageSmoothingQuality = 'high';
  ctx.drawImage(bitmap, 0, 0, size.w, size.h);
  return encodeWithFallback(encoder(canvas), QUALITY);
}

/** Przerysowanie na canvas usuwa EXIF (w tym GPS). Orientacja z EXIF jest uwzględniana przez createImageBitmap. */
export async function resizeImage(file: Blob): Promise<{ image: Blob; thumb: Blob }> {
  let bitmap: ImageBitmap;
  try {
    bitmap = await createImageBitmap(file, { imageOrientation: 'from-image' });
  } catch {
    throw new UnsupportedImageError();
  }
  try {
    return { image: await render(bitmap, MAX_SIDE), thumb: await render(bitmap, THUMB_SIDE) };
  } finally {
    bitmap.close();
  }
}

import { computed, signal } from '@angular/core';
import { UploadedImage } from '../../core/models';

export interface QueueItem {
  id: number;
  file: File;
  status: 'pending' | 'working' | 'done' | 'error';
  progress: number;
  result?: UploadedImage;
  error?: string;
}

type Processor = (file: File, onProgress: (pct: number) => void) => Promise<UploadedImage>;

export class UploadQueue {
  readonly items = signal<QueueItem[]>([]);
  readonly busy = computed(() => this.items().some((i) => i.status === 'pending' || i.status === 'working'));
  onDone: (item: QueueItem & { result: UploadedImage }) => void = () => undefined;
  private nextId = 1;
  private running = 0;

  constructor(private readonly process: Processor, private readonly concurrency = 3) {}

  add(files: File[], limit: number): { accepted: number; rejected: number } {
    const accepted = files.slice(0, Math.max(0, limit));
    this.items.update((list) => [
      ...list,
      ...accepted.map((file) => ({ id: this.nextId++, file, status: 'pending' as const, progress: 0 })),
    ]);
    this.pump();
    return { accepted: accepted.length, rejected: files.length - accepted.length };
  }

  retry(id: number): void {
    this.patch(id, { status: 'pending', progress: 0, error: undefined });
    this.pump();
  }

  remove(id: number): void {
    this.items.update((list) => list.filter((i) => i.id !== id || i.status === 'working'));
  }

  private pump(): void {
    while (this.running < this.concurrency) {
      const next = this.items().find((i) => i.status === 'pending');
      if (!next) return;
      this.running++;
      this.patch(next.id, { status: 'working' });
      this.process(next.file, (progress) => this.patch(next.id, { progress }))
        .then((result) => {
          this.patch(next.id, { status: 'done', progress: 100, result });
          this.onDone({ ...this.items().find((i) => i.id === next.id)!, result });
        })
        .catch((e: unknown) => this.patch(next.id, { status: 'error', error: e instanceof Error ? e.message : 'Nie udało się wysłać zdjęcia.' }))
        .finally(() => {
          this.running--;
          this.pump();
        });
    }
  }

  private patch(id: number, changes: Partial<QueueItem>): void {
    this.items.update((list) => list.map((i) => (i.id === id ? { ...i, ...changes } : i)));
  }
}

import { isPlatformBrowser } from '@angular/common';
import { inject, Injectable, OnDestroy, PLATFORM_ID } from '@angular/core';
import { resizeImage } from './image-resize';

type Result = { image: Blob; thumb: Blob };
type Reply = { id: number; ok: true; image: Blob; thumb: Blob } | { id: number; ok: false; error: string };

@Injectable({ providedIn: 'root' })
export class ImageResizer implements OnDestroy {
  private readonly browser = isPlatformBrowser(inject(PLATFORM_ID));
  private worker: Worker | null = null;
  private seq = 0;
  private readonly pending = new Map<number, { resolve: (r: Result) => void; reject: (e: Error) => void }>();

  resize(file: File): Promise<Result> {
    const worker = this.getWorker();
    if (!worker) return resizeImage(file);
    const id = ++this.seq;
    return new Promise<Result>((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      worker.postMessage({ id, file });
    });
  }

  ngOnDestroy(): void {
    this.worker?.terminate();
  }

  private getWorker(): Worker | null {
    if (this.worker) return this.worker;
    if (!this.browser || typeof Worker === 'undefined' || typeof OffscreenCanvas === 'undefined') return null;
    this.worker = new Worker(new URL('./image-resize.worker', import.meta.url), { type: 'module' });
    this.worker.onmessage = ({ data }: MessageEvent<Reply>) => {
      const p = this.pending.get(data.id);
      if (!p) return;
      this.pending.delete(data.id);
      if (data.ok) p.resolve({ image: data.image, thumb: data.thumb });
      else p.reject(new Error(data.error));
    };
    return this.worker;
  }
}

    effect(() => {
      const id = this.id();
      if (!id) return;
      this.api.post(id).subscribe({
        next: (p) => {
          this.form.setValue({ title: p.title, excerpt: p.excerpt, body: p.body, news: p.targets.news, groupIds: p.targets.groupIds, status: p.status });
          this.images.set(p.images);

        this.messages.add({ severity: 'success', summary: 'Wpis zapisany' });
        void this.router.navigate(['/admin/wpisy']);

      error: (e) => {
        this.saving.set(false);
        this.messages.add({ severity: 'error', summary: apiErrorMessage(e), life: 8000 });
      },
    });

    const id = this.id();
    if (!id) return;
    this.confirm.confirm({
      header: 'Usunąć wpis?',
      message: 'Wpis i jego zdjęcia zostaną trwale usunięte.',

      rejectLabel: 'Anuluj',

        next: () => void this.router.navigate(['/admin/wpisy']),
        error: (e) => this.messages.add({ severity: 'error', summary: apiErrorMessage(e) }),
      }),
    });
  }

import { Palette } from '../core/models';

function channel(c: number): number {
  const s = c / 255;
  return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4);
}

function luminance(hex: string): number {
  const n = parseInt(hex.slice(1), 16);
  return 0.2126 * channel((n >> 16) & 255) + 0.7152 * channel((n >> 8) & 255) + 0.0722 * channel(n & 255);
}

export function contrastRatio(a: string, b: string): number {
  const [l1, l2] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (l1 + 0.05) / (l2 + 0.05);
}

export function paletteWarnings(p: Palette): string[] {
  const checks: [string, string, string][] = [
    ['Przyciski (odcień 700 z białym tekstem)', p['700'], '#ffffff'],
    ['Nagłówki (kolor tekstu na białym tle)', p.text, '#ffffff'],
    ['Kafelki (kolor tekstu na odcieniu 100)', p.text, p['100']],
  ];
  return checks
    .filter(([, fg, bg]) => contrastRatio(fg, bg) < 4.5)
    .map(([label, fg, bg]) => `${label}: kontrast ${contrastRatio(fg, bg).toFixed(2)}:1 — wymagane min. 4.5:1`);
}
